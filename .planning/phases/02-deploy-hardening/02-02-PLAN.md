Replace everything from `echo "[4/4] Waiting for API health..."` through the end of the script with:

```bash
            # --- Helper: dump diagnostics for all services ---
            dump_diagnostics() {
              echo "--- docker compose ps ---"
              docker compose -f "${COMPOSE_FILE}" --env-file .env --env-file .env.cicd ps
              for svc in api admin-ui postgres rabbitmq; do
                echo "--- ${svc} logs (last 30 lines) ---"
                docker compose -f "${COMPOSE_FILE}" --env-file .env --env-file .env.cicd logs --tail=30 "${svc}" || true
              done
            }

            echo "[4/5] Waiting for API health at ${STAGE_HEALTHCHECK_URL} (max 3 min)..."
            API_HEALTHY=false
            for i in $(seq 1 36); do
              STATUS=$(curl -o /dev/null -s -w "%{http_code}" "${STAGE_HEALTHCHECK_URL}" 2>/dev/null || echo "000")
              echo "  attempt ${i}/36 — HTTP ${STATUS}"
              if [ "${STATUS}" = "200" ]; then
                echo "[4/5] API health passed"
                API_HEALTHY=true
                break
              fi
              sleep 5
            done

            if [ "${API_HEALTHY}" = "false" ]; then
              echo "[4/5] API health check failed after 3 min"
              dump_diagnostics
              exit 1
            fi

            echo "[5/5] Checking admin-ui..."
            ADMIN_UI_URL="${STAGE_ADMIN_UI_URL:-http://localhost:3010}"
            ADMIN_STATUS=$(curl -o /dev/null -s -w "%{http_code}" "${ADMIN_UI_URL}" 2>/dev/null || echo "000")
            echo "  admin-ui HTTP ${ADMIN_STATUS}"
            if [ "${ADMIN_STATUS}" = "200" ]; then
              echo "[5/5] Admin-UI check passed"
            else
              echo "[5/5] Admin-UI check failed — HTTP ${ADMIN_STATUS}"
              dump_diagnostics
              exit 1
            fi
```

Key changes from current script:
1. Steps renumbered from [4/4] to [4/5] and [5/5] (per D-02: also verify admin-ui)
2. Per D-03: On failure, `dump_diagnostics` outputs last 30 lines of ALL four service logs (api, admin-ui, postgres, rabbitmq) plus `docker compose ps` — not just api logs
3. Per D-02: After API health passes, a separate check verifies admin-ui responds HTTP 200
4. Default `STAGE_ADMIN_UI_URL` is `http://localhost:3010` (matching project convention, port 3010 on VPS)
5. The `dump_diagnostics` function is reused by both failure paths to keep the script DRY
6. No auto-rollback, no external alerts (per D-03: just fail + logs)
