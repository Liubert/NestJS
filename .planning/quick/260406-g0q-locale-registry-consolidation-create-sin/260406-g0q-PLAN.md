**1. Create `admin-ui/src/hooks/useSupportedLocales.ts`** (per Decision 6)

Create a React Query hook that fetches from `GET /translations/supported-locales`:
```ts
import { useQuery } from '@tanstack/react-query';
import apiClient from '../api/client';

export interface LocaleDefinition {
  code: string;
  name: string;
  aliases: string[];
  localeSkill: string;
  flag: string;
}

export function useSupportedLocales() {
  return useQuery<LocaleDefinition[]>({
    queryKey: ['supported-locales'],
    queryFn: () => apiClient.get('/translations/supported-locales').then(r => r.data),
    staleTime: 24 * 60 * 60 * 1000, // 24h — locale list rarely changes
  });
}

// Derived helpers
export function useLocaleMap() {
  const { data } = useSupportedLocales();
  const map = new Map<string, LocaleDefinition>();
  if (data) {
    for (const l of data) {
      map.set(l.code, l);
    }
  }
  return map;
}

export function useFlagForCode() {
  const map = useLocaleMap();
  return (code: string) => map.get(code)?.flag ?? '';
}
```
